3. احذف الكود الموجود والصق الكود التالي:

function doPost(e) {
  try {
    var sh = SpreadsheetApp.getActiveSheet();
    var raw = (e && e.postData && e.postData.contents) ? e.postData.contents : "{}";
    var d = JSON.parse(raw);

    var HEAD = ["Order ID","Date","Customer Name","Phone 1","Phone 2",
      "Wilaya","Commune","Delivery Type","Address","Product SKU",
      "Product Name","Quantity","Unit Price (DZD)","Total Amount (DZD)",
      "Confirmation","Notes"];

    if (sh.getLastRow() === 0) {
      sh.appendRow(HEAD);
      sh.getRange(1, 1, 1, HEAD.length).setFontWeight("bold")
        .setBackground("#0f9d58").setFontColor("#ffffff");
      sh.setFrozenRows(1);
    }

    // منع تكرار نفس الطلب عند إعادة التصدير
    var id = d.order_id || "";
    if (id && sh.getLastRow() > 1) {
      var col = sh.getRange(2, 1, sh.getLastRow() - 1, 1).getValues();
      for (var i = 0; i < col.length; i++) {
        if (String(col[i][0]) === String(id)) {
          return ContentService.createTextOutput("duplicate");
        }
      }
    }

    sh.appendRow([
      id,
      d.order_date || new Date(),
      d.customer_name || "",
      "'" + (d.phone_1 || ""),        // ' تمنع حذف الصفر الأول
      d.phone_2 ? "'" + d.phone_2 : "",
      d.wilaya || "",
      d.commune || "",
      d.delivery_type || "",
      d.address || "",
      d.product_sku || "",
      d.product_name || "",
      d.quantity || 1,
      d.unit_price || 0,
      d.total_amount || 0,
      d.confirmation || "قيد الانتظار",
      d.notes || ""
    ]);

    sh.autoResizeColumns(1, HEAD.length);
    return ContentService.createTextOutput("ok");
  } catch (err) {
    return ContentService.createTextOutput("error: " + err);
  }
}

function doGet() {                      // للتأكد أن النشر يعمل
